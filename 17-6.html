// 把 setTimeout 包装成 Promise
function wait(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

// 🥟 包饺子完整流程
async function makeDumplings() {
    const start = performance.now();

    // ① 串行：买完 → 取完
    await wait(6000);   // 🛒 买原料
    await wait(1000);   // 📦 取原料

    // ② 并行：FORK → 同时和面 & 和馅
    await Promise.all([
        wait(5000),  // 🥣 和面
        wait(3000)   // 🥕 和馅
    ]);  // JOIN —— 取 max(5000, 3000)

    // ③ 串行：包完 → 吃完 → 完成
    await wait(4000);   // 🥟 包饺子
    await wait(2000);   // 😋 吃饺子

    return performance.now() - start;
}
