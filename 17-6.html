// setTimeout → Promise
function wait(ms) { return new Promise(r => setTimeout(r, ms)); }

async function makeDumplings() {
    // ① Promise.race —— 谁先买到原料就用谁！
    const bought = await Promise.race([
        wait(6000).then(() => '线上'),
        wait(4000).then(() => '线下'),
    ]);  // 🏁 取最快的 —— 线下 4s 先完成！

    // ② 串行：取原料
    await wait(1000);

    // ③ Promise.all —— 两个都要做完才往下
    await Promise.all([
        wait(5000),  // 和面
        wait(3000),   // 和馅
    ]);  // 🎯 取最慢的 —— 和面 5s！

    // ④ 串行：包饺子 → 煮 → 吃 → 完
    await wait(4000);  // 🥟 包饺子
    await wait(3000);  // 🍲 煮饺子
    await wait(2000);  // 😋 吃饺子
}
