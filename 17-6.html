// 把 setTimeout 包装成 Promise
function wait(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

// 包饺子完整流程
async function makeDumplings() {
    const start = performance.now();

    // 串行：买完 → 取完
    await wait(6000);   // 🛒 买原料
    await wait(1000);   // 📦 取原料

    // 并行：同时和面 & 和馅（Promise.all）
    await Promise.all([
        wait(5000),  // 🥣 和面
        wait(3000)   // 🥕 和馅
    ]);

    // 串行：包完 → 吃完 → 完成
    await wait(4000);   // 🥟 包饺子
    await wait(2000);   // 😋 吃饺子

    return performance.now() - start;
}
